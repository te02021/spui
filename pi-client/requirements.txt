# SPUI Pi Client — dependencias Python
# Instalar con: pip install -r requirements.txt

# HTTP — sync REST con el CMS
requests>=2.28.0

# MQTT — alertas de emergencia y telemetría en tiempo real
# El código soporta tanto la API 1.x como la 2.x de paho (ver mqtt_listener.py).
paho-mqtt>=1.6.1

# VLC — reproductor multimedia
# Requiere VLC instalado en el sistema:
#   Pi/Linux: sudo apt install vlc
#   Windows:  https://www.videolan.org/vlc/
#
# El esquema de versiones de python-vlc es 3.0.<libvlc><build>, NO una fecha:
# 3.0.18121, 3.0.20123, 3.0.21203... Se fija la serie 3.x porque es la que
# corresponde a VLC 3.x, que es el que trae Raspberry Pi OS Bookworm.
python-vlc>=3.0.18121,<4

# Telemetría — métricas del sistema (CPU temp, RAM, disco)
psutil>=5.9.0

# Imágenes — dibuja cada contenido con su plantilla y su tema (spui/tv.py), la
# pantalla de espera y la de alerta. Sin Pillow el cliente cae al texto por
# overlay, que funciona pero es menos confiable. Viene preinstalado en
# Raspberry Pi OS. 9.2 trae rounded_rectangle y getlength.
Pillow>=9.2.0

# Ventana del TV (spui/ventana.py, noviembre 2026): muestra lo que dibuja
# spui/tv.py y desliza las páginas del cronograma; VLC dibuja el video dentro
# de ella. pygame-ce y no pygame: es la rama mantenida, con ruedas para la Pi
# (trae su propio SDL2, no hace falta nada de apt) y se importa igual
# (`import pygame`). No instalar los dos juntos: se pisan.
pygame-ce>=2.4
